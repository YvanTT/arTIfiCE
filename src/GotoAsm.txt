#define curRow					$D00595
#define curPC					$D0231A
#define endPC					$D0231D

	.assume adl = 1																; code is in adl mode

.org 0

	.db $BB,$0F,$EF,$97,$BB,$16,$31,$31,$35,$2B,$30,$11,$11,$70,$2A,$AF			; :inString(toString(binomcdf(115,0))+"?","?")

__asm_start:
	; ASM payload to execute. Note: Non-fixed address.
	; hl = PC - 2
